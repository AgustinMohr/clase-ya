# ----------------------------------------------------------------------------
# 1) BASE DE DATOS LOCAL (Postgres en Docker)
# ----------------------------------------------------------------------------
# Primera vez (crea el contenedor):
docker run --name claseya-pg -e POSTGRES_USER=claseya -e POSTGRES_PASSWORD=claseya -e POSTGRES_DB=claseya -p 5432:5432 -d postgres:16

docker start claseya-pg        # siguientes veces
docker stop claseya-pg         # al terminar
docker logs claseya-pg         # si algo no conecta

# ----------------------------------------------------------------------------
# 2) BACKEND (:8080)
# ----------------------------------------------------------------------------
# IMPORTANTE: que DB_URL / DB_USERNAME / DB_PASSWORD NO esten seteadas en la terminal.
# Si lo estan, el backend local escribe en NEON (PRODUCCION). Verificar y limpiar:
$env:DB_URL; $env:DB_USERNAME; $env:DB_PASSWORD
Remove-Item Env:DB_URL, Env:DB_USERNAME, Env:DB_PASSWORD -ErrorAction SilentlyContinue

# Levantar con el perfil local (el timezone es obligatorio en Windows, ver seccion 7):
mvn spring-boot:run "-Dspring-boot.run.profiles=local" "-Dspring-boot.run.jvmArguments=-Duser.timezone=UTC"

# CON DEVTOOLS: al recompilar, la app se reinicia sola -> NO hace falta cortar el proceso
# ni volver a levantarlo cuando cambias codigo Java. Alcanza con guardar en VS Code
# (la extension de Java recompila) o correr `mvn compile` en otra terminal.
# Devtools es dev-only: no entra en el jar y no afecta el deploy.

# Variante sin tests (arranque mas rapido en el loop diario):
mvn -DskipTests spring-boot:run "-Dspring-boot.run.profiles=local" "-Dspring-boot.run.jvmArguments=-Duser.timezone=UTC"

# En el arranque tiene que decir:  The following 1 profile is active: "local"
# Health check y datos (el 401 de "/" es ESPERADO: "/" no es un endpoint):
#   http://localhost:8080/api/test/public
#   http://localhost:8080/api/subjects?size=5
#   http://localhost:8080/api/teachers?page=0&size=24

# ----------------------------------------------------------------------------
# 3) FRONTEND (:5173, proxy /api -> :8080)
# ----------------------------------------------------------------------------
cd frontend
npm install                    # primera vez
npm run dev                    # http://localhost:5173
# En local NO se define VITE_API_URL: api.ts usa URL relativa y el proxy de Vite
# manda /api a :8080, asi que en local NO hay CORS.

# ----------------------------------------------------------------------------
# 4) TESTS Y VERIFICACION
# ----------------------------------------------------------------------------
mvn -B clean test              # suite completa (unit + integracion; requiere Docker)
scripts\test.ps1               # preflight de Docker + mvn -B clean test
scripts\verify.ps1             # verificacion reproducible (reporta PASS/FAIL)
# desde la raiz:
cd frontend; npm run build     # tsc + vite: atrapa errores de tipos

# ---- E2E (Playwright) -------------------------------------------------------
# Levanta su PROPIO backend (perfil local) + el dev server de Vite y prueba los
# flujos criticos contra la base demo local. Requiere Docker y `claseya-pg`.
# Primera vez: npx playwright install chromium
scripts\e2e.ps1                # backend propio (determinista)
scripts\e2e.ps1 -ReuseBackend  # rapido: reusa el backend ya levantado (solo si tiene
                               # el codigo actual compilado)
# Al fallar, el reporte queda en frontend\test-results\<test>\ (screenshot +
# error-context.md) y el trace se puede abrir con: npx playwright show-trace

# ----------------------------------------------------------------------------
# 6) DEPLOY (Render backend + Vercel frontend + Neon)
# ----------------------------------------------------------------------------
# Render (Web Service, Language Docker, Dockerfile ./Dockerfile,
#         Health Check Path /api/test/public). Env vars:
#   DB_URL=jdbc:postgresql://<host-neon>/neondb?sslmode=require&prepareThreshold=0
#   DB_USERNAME=neondb_owner
#   DB_PASSWORD=<password de Neon>
#   JWT_SECRET=<aleatorio, >=32 caracteres>   JWT_EXPIRATION=3600
#     -> OBLIGATORIO en produccion: el backend NO arranca con el secret de desarrollo ni
#        con uno de menos de 32 bytes fuera del perfil `local` (fail-fast al iniciar).
#   GOOGLE_CLIENT_ID=<EL MISMO que VITE_GOOGLE_CLIENT_ID>
#   FRONTEND_URL=https://<dominio-vercel>    (origin exacto, SIN barra final)
#   DEMO_SEED=false                          (en produccion real)
#
# Vercel: Root Directory=frontend | build "npm run build" | output "dist"
#   VITE_API_URL=https://<dominio-render>    (SIN barra final)
#   VITE_GOOGLE_CLIENT_ID=<client id>
#   -> cambiar una VITE_* exige REDEPLOY: Vite las incrusta durante el build.
#
# Google Cloud > Credentials > OAuth Client > Authorized JavaScript origins:
#   http://localhost:5173  y  https://<dominio-vercel>   (sin path, sin wildcard)
#   Los preview URLs de Vercel son origins distintos: no estan autorizados.
